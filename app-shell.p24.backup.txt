function initials(name: string) {
  return name
    .replace(/(Pvt\.?|Ltd\.?|Services|Partners)/gi, "")
    .trim()
    .split(/\s+/)
    .slice(0, 2)
    .map((w) => w[0])
    .join("")
    .toUpperCase();
}

export function BranchSelector() {
  const { branch, setBranch, tenantBranches, canUseAllBranches, tenant } = useApp();
  const currentValue =
    branch !== "all" && tenantBranches.some((b) => b.id === branch)
      ? branch
      : canUseAllBranches
        ? "all"
        : (tenantBranches[0]?.id ?? "all");
  return (
    <Select value={currentValue} onValueChange={setBranch}>
      <SelectTrigger className="w-[190px] border-border bg-card shadow-sm" aria-label="Branch context">
        <span className="text-[9px] font-semibold uppercase tracking-widest text-muted-foreground">
          Branch&nbsp;
        </span>
        <SelectValue />
      </SelectTrigger>
      <SelectContent>
        {(canUseAllBranches || tenantBranches.length === 0) && <SelectItem value="all">All Branches</SelectItem>}
        {tenantBranches.map((b) => (
          <SelectItem key={b.id} value={b.id} disabled={b.status === "Inactive" && !canUseAllBranches}>
            {b.name}
            <span className="ml-2 text-xs text-muted-foreground">{b.code}</span>
          </SelectItem>
        ))}
        {!canUseAllBranches && tenant.role === "Field Officer" && (
          <SelectItem value="__none" disabled className="text-xs text-muted-foreground">
            Only your assigned branches
          </SelectItem>
        )}
      </SelectContent>
    </Select>
  );
}

function NavItem({ item, pathname }: { item: { label: string; to: string; icon: LucideIcon }; pathname: string }) {
  const active = pathname === item.to || (item.to !== "/" && pathname.startsWith(item.to));
  return (
    <Link
      to={item.to}
      className={cn(
        "group relative flex items-center gap-2.5 rounded-lg px-3 py-2 text-[13px] font-medium transition-colors",
        active
          ? "bg-sidebar-accent text-sidebar-accent-foreground"
          : "text-sidebar-foreground/75 hover:bg-sidebar-accent/60 hover:text-sidebar-foreground",
      )}
    >
      {active && <span className="absolute left-0 top-1/2 h-5 w-1 -translate-y-1/2 rounded-r-full bg-accent" />}
      <item.icon className={cn("size-4 shrink-0", active ? "text-accent" : "text-sidebar-foreground/60")} />
      {item.label}
    </Link>
  );
}
