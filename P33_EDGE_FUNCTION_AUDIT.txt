        hint: 'If this identity exists already, use Link Existing Auth User in ValuationPro.' });

    return result(200, { user_id: String(id), invited: true });
  } catch (e) {
    return result(500, { error: e instanceof Error ? e.message : 'Invitation failed' });

type Branch = { id: string; name: string; code: string; is_active: boolean };

export type ProvisionResult = { name: string; email: string; role: string;
  status: 'created' | 'failed' | 'reset'; user_id?: string;
  temporary_password?: string; error?: string; warning?: string };
type ImportEntry = NewUser & { excel_row: number; error: string | null };
const emailValid = (email: string) => /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email) && email.length <= 320;
function errorText(e: unknown): string { return e instanceof Error ? e.message : String(e); }
async function invokeProvision(body: Record<string, unknown>): Promise<{ results: ProvisionResult[] }> {
  const { data, error } = await supabase.functions.invoke('vp-provision-users', { body });
  if (error) {
    let message = error.message;
    const context = (error as any).context;
    if (context && typeof context.json === 'function') {
      try { const json = await context.json(); message = json.error ?? message; } catch { /* ignore */ }
    }
    throw new Error(message);
  }
  if (data?.error) throw new Error(String(data.error));
  return data as { results: ProvisionResult[] };
}
function csvCell(value: string) {
  const neutral = /^[=+\-@\t\r]/.test(value) ? `'${value}` : value;
  return `"${neutral.replaceAll('"', '""')}"`;
}
function downloadCredentials(items: ProvisionResult[], fileName: string) {
  const headers = ['Full Name', 'Email', 'Role', 'Temporary Password', 'Status', 'Error'];
  const rows = items.map(x => [x.name, x.email, x.role, x.temporary_password ?? '', x.status, x.error ?? x.warning ?? '']);
  const content = '\uFEFF' + [headers, ...rows].map(row => row.map(cell => csvCell(String(cell))).join(',')).join('\r\n');
  const url = URL.createObjectURL(new Blob([content], { type: 'text/csv;charset=utf-8' }));
  const link = document.createElement('a'); link.href = url; link.download = fileName; link.click();
  setTimeout(() => URL.revokeObjectURL(url), 10_000);
}

export function VpUserProvisionDialog({ open, onOpenChange, roles, branches, onComplete }: {
  open: boolean; onOpenChange: (open: boolean) => void;
  roles: string[]; branches: Branch[]; onComplete: () => Promise<void>;
}) {
  const [mode, setMode] = useState<'single' | 'bulk' | 'link'>('single');
  const [user, setUser] = useState<NewUser>({ name: '', email: '', role: '', branch_codes: [] });
  const [rows, setRows] = useState<ImportEntry[]>([]);
  const [busy, setBusy] = useState(false);
  const [progress, setProgress] = useState(0);
  const [results, setResults] = useState<ProvisionResult[] | null>(null);
  const activeCodes = useMemo(() => new Set(branches.filter(x => x.is_active).map(x => x.code.toUpperCase())), [branches]);
  const validRows = rows.filter(r => !r.error);
  const failedRows = rows.filter(r => r.error);
  const isValid = (x: NewUser): string | null => {
    if (x.name.trim().length < 2 || x.name.length > 200) return 'Name must be 2â€“200 characters';
    if (!emailValid(x.email)) return 'Invalid email';
    if (!roles.includes(x.role)) return `Unknown role: ${x.role}`;
    const unknown = x.branch_codes.find(code => !activeCodes.has(code));
    return unknown ? `Unknown/inactive branch: ${unknown}` : null;
  };
  async function readExcel(event: ChangeEvent<HTMLInputElement>) {
    const file = event.target.files?.[0]; if (!file) return;
    setResults(null); setRows([]); setProgress(0);
    if (!/\.xlsx$/i.test(file.name) || file.size > 3_000_000) {
      toast.error('Select a .xlsx file up to 3 MB'); return;
    }
    try {
      const { readSheet } = await import('read-excel-file/browser');
      const data = await readSheet(file);
      const headers = (data[0] ?? []).map(v => String(v ?? '').trim().toLowerCase());
      const expected = ['full name', 'email', 'role', 'branch codes'];
      if (expected.some((name, idx) => headers[idx] !== name))
        throw new Error('Excel columns must be: Full Name | Email | Role | Branch Codes');
      const records = data.slice(1).filter(line => line.some(cell => String(cell ?? '').trim()));
      if (!records.length || records.length > 1000) throw new Error('Workbook must contain 1â€“1000 users');
      const seen = new Set<string>();
      setRows(records.map((cells, index) => {
        const row: NewUser = {
          name: String(cells[0] ?? '').trim(), email: String(cells[1] ?? '').trim().toLowerCase(),
          role: String(cells[2] ?? '').trim(),
          branch_codes: Array.from(new Set(String(cells[3] ?? '').split(';').map(s => s.trim().toUpperCase()).filter(Boolean))),
        };
        let error = isValid(row);
        if (seen.has(row.email)) error = 'Repeated email in import';
        seen.add(row.email);
        return { ...row, excel_row: index + 2, error };
      }));
    } catch (e) { toast.error(errorText(e)); }
  }
  async function provision() {
    if (busy) return;
    setBusy(true); setProgress(0); setResults(null);
    try {
      if (mode === 'link') {
        const error = isValid(user); if (error) throw new Error(error);
        const { data: id, error: linkError } = await vpDb.rpc('vp_admin_attach_invited_user', {
          p_name: user.name.trim(), p_email: user.email.trim().toLowerCase(), p_role: user.role,
        });
        if (linkError) throw linkError;
        if (user.branch_codes.length) {
          const ids = user.branch_codes.map(code => branches.find(b => b.code.toUpperCase() === code)?.id).filter(Boolean);
          const { error: branchError } = await vpDb.rpc('vp_admin_replace_user_branches', {
            p_user_id: String(id), p_branch_ids: ids,
          });
          if (branchError) toast.warning('Linked account; review branch assignments');
        }
        toast.success('Existing login linked to this agency');
        await onComplete(); onOpenChange(false);
      } else {
        const batch = mode === 'single' ? [user] : validRows;
        if (mode === 'single') { const error = isValid(user); if (error) throw new Error(error); }
        if (!batch.length) throw new Error('No valid users to create');
        const outcomes: ProvisionResult[] = [];
        for (let i = 0; i < batch.length; i += 20) {
          const part = batch.slice(i, i + 20);
          try {
            const data = await invokeProvision(mode === 'single' ? { action: 'create', user: part[0] }
              : { action: 'bulk', rows: part });
            outcomes.push(...data.results);
          } catch (e) {
            outcomes.push(...part.map(x => ({ ...x, status: 'failed' as const, error: errorText(e) })));
          }
          setProgress(Math.min(i + part.length, batch.length));
        }
        outcomes.push(...failedRows.map(x => ({ ...x, status: 'failed' as const, error: `Row ${x.excel_row}: ${x.error}` })));
        setResults(outcomes);
        await onComplete();
        const successes = outcomes.filter(x => x.status === 'created').length;
        toast[successes ? 'success' : 'error'](`${successes} user(s) created; ${outcomes.length - successes} failed`);
      }
    } catch (e) { toast.error(errorText(e)); }
    finally { setBusy(false); }
  }
  function close() {
    if (busy) return;
    setResults(null); setRows([]); setProgress(0);
    onOpenChange(false);
  }
  return <>
    <Dialog open={open && !results} onOpenChange={v => { if (!v && !results) close(); else if (v) onOpenChange(true); }}>
      <DialogContent className="sm:max-w-2xl">
        <DialogHeader><DialogTitle>User Management â€” Add Accounts</DialogTitle>
          <DialogDescription>Create staff logins directly. No email invitations or database-console steps are needed.</DialogDescription>
        </DialogHeader>
        <div className="flex flex-wrap gap-2">
          <Button type="button" size="sm" variant={mode === 'single' ? 'default' : 'outline'} disabled={busy} onClick={() => setMode('single')}>
            <UserRoundPlus className="size-4" /> Create User</Button>
          <Button type="button" size="sm" variant={mode === 'bulk' ? 'default' : 'outline'} disabled={busy} onClick={() => setMode('bulk')}>
            <Upload className="size-4" /> Excel Bulk Import</Button>
          <Button type="button" size="sm" variant={mode === 'link' ? 'default' : 'outline'} disabled={busy} onClick={() => setMode('link')}>
            <Plus className="size-4" /> Link Existing Account</Button>
        </div>
        {mode !== 'bulk' ? <div className="space-y-4">
          <div><Label>Full name</Label><Input value={user.name} onChange={e => setUser(x => ({ ...x, name: e.target.value }))} placeholder="Employee full name" /></div>
          <div><Label>Work email / login</Label><Input type="email" value={user.email} onChange={e => setUser(x => ({ ...x, email: e.target.value }))} placeholder="staff@company.com" /></div>
          <div><Label>Role</Label><Select value={user.role || undefined} onValueChange={role => setUser(x => ({ ...x, role }))}>
            <SelectTrigger><SelectValue placeholder="Choose a role" /></SelectTrigger>
            <SelectContent>{roles.map(role => <SelectItem key={role} value={role}>{role}</SelectItem>)}</SelectContent>
          </Select>
          {!roles.length && <p role="alert" className="mt-1 text-xs text-destructive">No assignable roles available. Ask the Tenant Owner to configure the Role Matrix.</p>}
          </div>
          <div><Label>Branch access (optional)</Label><div className="mt-2 flex max-h-28 flex-wrap gap-3 overflow-auto rounded border p-3">
            {branches.filter(b => b.is_active).map(branch => <label className="flex items-center gap-2 text-xs" key={branch.id}>
              <Checkbox checked={user.branch_codes.includes(branch.code)} onCheckedChange={checked =>
                setUser(x => ({ ...x, branch_codes: checked === true ? [...x.branch_codes, branch.code]
                  : x.branch_codes.filter(code => code !== branch.code) }))} />

export async function resetValuationUser(userId: string): Promise<ProvisionResult> {
  const { data, error } = await supabase.functions.invoke('vp-provision-users', {
    body: { action: 'reset_password', user_id: userId },
  });
  if (error) {
    let message = error.message;
    const context = (error as any).context;
    if (context && typeof context.json === 'function') {
      try { const obj = await context.json(); message = obj.error ?? message; } catch { /* ignore */ }
    }
    throw new Error(message);
  }
  if (data?.error) throw new Error(data.error);
  return data as ProvisionResult;
}

import { useEffect, useRef, useState, type FormEvent, type ReactNode } from 'react';

type Gate = 'loading' | 'login' | 'setup' | 'blocked' | 'ready' | 'password-setup' | 'error';

export function VpAuthGate({ children }: { children: ReactNode }) {
  const pathname = useRouterState({ select: (r) => r.location.pathname });
  const navigate = useNavigate();
  const queryClient = useQueryClient();
  const [gate, setGate] = useState<Gate>('loading');
  const refreshSequence = useRef(0);
  const [identity, setIdentity] = useState<VpIdentity | null>(null);
  const [capabilities, setCapabilities] = useState<string[]>([]);
  const [error, setError] = useState('');
  const [notice, setNotice] = useState('');
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [register, setRegister] = useState(false);
  const [name, setName] = useState('');
  const [code, setCode] = useState('');
  const [busy, setBusy] = useState(false);
  const [serverPasswordRequired, setServerPasswordRequired] = useState(false);
  const [invitePassword, setInvitePassword] = useState('');
  const [confirmInvitePassword, setConfirmInvitePassword] = useState('');

  async function refresh() {
    const currentSequence = ++refreshSequence.current;
    const stillCurrent = () => currentSequence === refreshSequence.current;
    try {
      const { data: { user }, error: authError } = await supabase.auth.getUser();
      if (!stillCurrent()) return;
      if (authError || !user) {
        queryClient.clear();
        setIdentity(null);
        setCapabilities([]);
        setGate('login');
        return;
      }
      const { data, error: rpcError } = await vpDb.rpc('vp_my_context');
      if (!stillCurrent()) return;
      if (rpcError) throw rpcError;
      const profile = data as VpIdentity | null;
      if (profile) {
        // Fail closed if the P10 database security migration is missing.
        const { data: granted, error: permissionsError } = await vpDb.rpc('vp_sec_my_capabilities');
        if (!stillCurrent()) return;
        if (permissionsError) throw new Error('P10 permissions are unavailable. Apply the P10 migration and retry: ' + permissionsError.message);
        if (!Array.isArray(granted)) throw new Error('Unexpected P10 permission response');
        setCapabilities(granted.map(String));
      } else {
        setCapabilities([]);
      }
      setIdentity(profile);
      if (profile) {
        const { data: firstLogin, error: setupError } = await vpDb.rpc('vp_first_login_required');
        if (!stillCurrent()) return;
        if (setupError) throw setupError;
        setServerPasswordRequired(firstLogin === true);
        const meta = user.user_metadata ?? {};
        setGate(firstLogin === true || (meta['vp_invited'] === true && meta['vp_password_set'] !== true)
          ? 'password-setup' : 'ready');
      }
      else {
        const { data: accountStatus, error: statusError } = await vpDb.rpc('vp_account_status');
        if (!stillCurrent()) return;
        if (statusError) throw statusError;
        setGate(accountStatus === 'unlinked' ? 'setup' : 'blocked');
      }
    } catch (e) {
      if (!stillCurrent()) return;
      setError(e instanceof Error ? e.message : 'Unable to verify the session');
      setGate('error');
    }
  }

  useEffect(() => {
    void refresh();
    const handlePermissionChange = () => { void refresh(); };
    window.addEventListener('vp-permissions-changed', handlePermissionChange);
    const { data: { subscription } } = supabase.auth.onAuthStateChange((event) => {
      if (event === 'SIGNED_OUT') {
        ++refreshSequence.current;
        queryClient.clear();
        setIdentity(null);
        setCapabilities([]);
        setGate('login');
      }
      // Don't await a Supabase query within the auth callback: the library
      // documents potential lock contention. Defer to a separate microtask.
      if (event === 'SIGNED_IN' || event === 'SIGNED_OUT' || event === 'USER_UPDATED') {
        queueMicrotask(() => { void refresh(); });
      }
    });
    return () => { subscription.unsubscribe(); window.removeEventListener('vp-permissions-changed', handlePermissionChange); };
  }, []);

  // A deep link must not become the login URL or a future post-login redirect.
  // The authenticated home screen is always /; modules are selected afterwards.
  useEffect(() => {
    if (gate === 'login' && pathname !== '/') {
      void navigate({ to: '/', replace: true });
    }
  }, [gate, pathname, navigate]);

  async function submitLogin(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    setError('');
    setNotice('');
    setBusy(true);
    try {
      if (register) {
        const { data, error: signUpError } = await supabase.auth.signUp({ email, password });
        if (signUpError) throw signUpError;
        if (!data.session) {
          setError('Check your email and confirm your account, then sign in.');
          setRegister(false);
        } else {
          await refresh();
          await navigate({ to: '/', replace: true });
        }
      } else {
        const { error: signInError } = await supabase.auth.signInWithPassword({ email, password });
        if (signInError) throw signInError;
        // Prevent data belonging to a previous login from appearing briefly.
        queryClient.clear();
        await refresh();
        await navigate({ to: '/', replace: true });
      }
    } catch (e) {
      setError(e instanceof Error ? e.message : 'Sign in failed');
    } finally { setBusy(false); }
  }

  async function createAgency(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    setError('');
    setBusy(true);
    try {
      const { error: rpcError } = await vpDb.rpc('vp_register_agency', {
        p_code: code.trim().toUpperCase(), p_name: name.trim(),
      });
      if (rpcError) throw rpcError;
      await refresh();
    } catch (e) { setError(e instanceof Error ? e.message : 'Agency setup failed'); }
    finally { setBusy(false); }
  }

  async function completeInvitePassword(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    if (invitePassword.length < 12 || !/[A-Za-z]/.test(invitePassword) || !/[0-9]/.test(invitePassword) || invitePassword !== confirmInvitePassword) {
      setError('Use a matching password of at least 12 characters containing letters and numbers'); return;
    }
    setBusy(true); setError('');
    try {
      if (serverPasswordRequired) {
        const { data, error: updateError } = await supabase.functions.invoke('vp-provision-users', {
          body: { action: 'set_password', password: invitePassword },
        });
        if (updateError) throw updateError;
        if (data?.error) throw new Error(data.error);
        setInvitePassword(''); setConfirmInvitePassword('');
        const { error: signOutError } = await supabase.auth.signOut();
        if (signOutError) throw signOutError;
        queryClient.clear();
        await navigate({ to: '/', replace: true });
        setIdentity(null);
        setCapabilities([]);
        setNotice('Password updated. Please sign in using your new password.');
        setGate('login');
        return;
      } else {
        const { error: updateError } = await supabase.auth.updateUser({
          password: invitePassword, data: { vp_password_set: true },
        });
        if (updateError) throw updateError;
      }
      setInvitePassword(''); setConfirmInvitePassword('');
      await refresh();
    } catch (e) { setError(e instanceof Error ? e.message : 'Password update failed'); }
    finally { setBusy(false); }
  }

  async function signOut() {
    try {
      setError('');
      await vpSignOutToHome(queryClient);
    } catch (error) {
      setError(error instanceof Error ? error.message : 'Sign out failed');
    }
  }

  if (gate === 'loading') {
    return <div className="flex min-h-screen items-center justify-center bg-background text-sm text-muted-foreground">Verifying ValuationPro sessionâ€¦</div>;

  if (gate === 'ready' && identity) {

Ask the administrator to update its Role Matrix.</p><button className="rounded border px-3 py-2" onClick={signOut}>Sign out</button></main>;
    }
    return <AppProvider identity={identity} capabilities={capabilities}>{children}</AppProvider>;
  }
  return (
    <div className="grid min-h-screen bg-background text-foreground lg:grid-cols-[1fr_0.9fr]">
      <section className="relative hidden overflow-hidden bg-[#0f2542] px-12 py-12 text-white lg:flex lg:flex-col lg:justify-center xl:px-20" aria-label="About ValuationPro">
        <div className="relative z-10 max-w-xl space-y-6"><div className="flex items-center gap-3"><Building2 className="size-8" aria-hidden="true"/><span className="text-xl font-semibold">ValuationPro</span></div>
          <h2 className="text-4xl font-semibold leading-tight tracking-tight xl:text-5xl">Property valuation.<br/>One connected workspace.</h2>
          <p className="max-w-md text-base leading-relaxed text-slate-200">Manage case intake, site inspections, valuation workpapers and quality review in one platform.</p>

// P10 role-matrix navigation. This is a UX guard only; PostgreSQL RLS/RPCs enforce data access.
// Administration (tenant.admin) is deliberately a separate recovery permission.
export function hasRolePermission(capabilities: readonly string[], action: string): boolean {
  return capabilities.includes(action);
}
export type AccessRule = { path: string; anyOf: readonly string[] };
export const accessRules: readonly AccessRule[] = [
  { path: '/admin/roles-permissions', anyOf: ['tenant.admin'] },
  { path: '/admin/users', anyOf: ['tenant.admin'] },
  { path: '/admin/', anyOf: ['tenant.admin'] },
  { path: '/cases/', anyOf: ['cases.View', 'field.View', 'field.Check In/Out', 'qc.View', 'signoff.View'] },
  { path: '/cases', anyOf: ['cases.View'] },
  { path: '/properties', anyOf: ['cases.View'] },
  { path: '/market-intelligence', anyOf: ['cases.View'] },
  { path: '/field-inspections/', anyOf: ['field.View', 'field.Check In/Out', 'qc.View'] },
  { path: '/field-operations', anyOf: ['field.View','field.Assign','field.Check In/Out'] },
  { path: '/qc-checking', anyOf: ['qc.View'] },
  { path: '/reports', anyOf: ['signoff.View', 'signoff.Generate'] },
  { path: '/billing-mis', anyOf: ['billing.View'] },
  { path: '/analytics', anyOf: ['analytics.View'] },
  { path: '/notifications', anyOf: [] }, // personal notifications are separately recipient-scoped.
  // Any authenticated member may load the safe home shell. Dashboard data
  // remains guarded by analytics.View in both React and the database RPC.
  { path: '/', anyOf: [] },
];
export function requiredPermissions(pathname: string): readonly string[] {
  const match = accessRules.find(r => pathname === r.path ||
    (r.path !== '/' && r.path.endsWith('/') && pathname.startsWith(r.path)));
  return match?.anyOf ?? [];
}
export function mayOpenPath(pathname: string, capabilities: readonly string[]): boolean {
  const required = requiredPermissions(pathname);
  // Unknown paths fail closed, except public onboarding which is handled by VpAuthGate.
  if (!accessRules.some(r => pathname === r.path || (r.path !== '/' && r.path.endsWith('/') && pathname.startsWith(r.path))))
    return pathname === '/onboarding';
  return required.length === 0 || required.some(x => hasRolePermission(capabilities, x));
}

export async function vpSignOutToHome(queryClient: QueryClient): Promise<void> {

          <p className="text-xs text-muted-foreground">For a Checker, grant qc.View, qc.Approve and qc.Return in Role Matrix. Only a different maker can be selected as its checker.</p>
        </div>}
        <DialogFooter><Button variant="outline" onClick={()=>setSelected(null)}>Cancel</Button>
          <Button disabled={busy} onClick={()=>void saveUser()}>Save changes</Button></DialogFooter>
      </DialogContent>
    </Dialog>
  </AppShell>;
}

import { describe, it, expect } from 'vitest';
import { mayOpenPath } from '@/lib/vp-p10-permissions';

describe('P10 role matrix navigation (backend must independently deny)', () => {
  it('denies operational modules when every checkbox is unchecked', () => {
    const adminRecoveryOnly = ['tenant.admin'];
    for (const path of ['/cases','/field-operations','/qc-checking','/reports','/billing-mis','/analytics'])
      expect(mayOpenPath(path, adminRecoveryOnly)).toBe(false);
    expect(mayOpenPath('/admin/users', adminRecoveryOnly)).toBe(true);
    expect(mayOpenPath('/admin/roles-permissions', adminRecoveryOnly)).toBe(true);
  });
  it('always allows the safe home route, without granting analytics data', () => {
    expect(mayOpenPath('/', [])).toBe(true);
    expect(mayOpenPath('/analytics', [])).toBe(false);
  });
  it('denies direct URLs, not just sidebar entries', () => {
    expect(mayOpenPath('/cases/121', [])).toBe(false);
    expect(mayOpenPath('/field-inspections/8', [])).toBe(false);
    expect(mayOpenPath('/admin/users', ['cases.View'])).toBe(false);
    expect(mayOpenPath('/unknown-private-page', ['cases.View'])).toBe(false);
  });
  it('allows specifically permitted modules and assigned field workflow routes', () => {
    expect(mayOpenPath('/cases', ['cases.View'])).toBe(true);
    expect(mayOpenPath('/cases/121', ['field.Check In/Out'])).toBe(true);
    expect(mayOpenPath('/field-inspections/8', ['field.Check In/Out'])).toBe(true);
    expect(mayOpenPath('/qc-checking', ['qc.View'])).toBe(true);
    expect(mayOpenPath('/billing-mis', ['cases.View'])).toBe(false);
  });
});
